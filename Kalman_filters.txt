//@version=5
strategy("Kalman Adaptive Scalper v1.6 — MCX Silver Micro",
     overlay=false,
     default_qty_type=strategy.percent_of_equity,
     default_qty_value=2,
     commission_type=strategy.commission.percent,
     commission_value=0.1,
     calc_on_every_tick=false,
     process_orders_on_close=true)

// ═══════════════════════════════════════════════════════
// INPUTS
// ═══════════════════════════════════════════════════════

// Kalman Filter
kalmQ   = input.float(0.004, "Kalman Process Noise (Q)",      group="Kalman Filter", step=0.001)
kalmR   = input.float(1.3,   "Kalman Measurement Noise (R)",  group="Kalman Filter", step=0.1)

// Entry
entryThresh  = input.float(1.8, "Entry Threshold (ATR multiples)", group="Entry", step=0.1)
exitThresh   = input.float(0.3, "Exit Threshold (ATR multiples)",  group="Entry", step=0.1)

// OU Half-Life Gate
ouLen       = input.int(19,  "OU Fit Window (bars)",      group="OU Gate")
maxHalfLife = input.int(90,  "Max Half-Life (bars)",      group="OU Gate")
useOuGate   = input.bool(true,"Enable OU Gate",           group="OU Gate")

// Gap Circuit Breaker
useGapBreaker = input.bool(true, "Enable Gap Breaker",          group="Gap Breaker")
gapThresh     = input.float(1.5, "Opening Gap Threshold (ATR)", group="Gap Breaker", step=0.1)
rangeThresh   = input.float(2.5, "Bar Range Threshold (ATR)",   group="Gap Breaker", step=0.1)
lockoutBars   = input.int(20,   "Lockout Period (bars)",        group="Gap Breaker")

// Regime filter — Variance Ratio
vrLen        = input.int(36,   "Variance Ratio Lookback",     group="Regime Filter")
vrThreshold  = input.float(0.7,"Max Variance Ratio (VR<1=MR)",group="Regime Filter", step=0.05)

// HURST R/S PROXY — NEW — acts as extreme-trending veto only
useHurstGate    = input.bool(true, "Enable Hurst Gate",        group="Hurst Filter")
hurstLen        = input.int(64, "Hurst Lookback",               group="Hurst Filter",
     tooltip="Bars used for R/S computation. Pine limits make 32-96 practical. 64 is standard.")
hurstMaxH       = input.float(0.60, "Max H (block if exceeded)",group="Hurst Filter", step=0.01,
     tooltip="Block entries when Hurst exponent exceeds this (strong trending). 0.5=random walk. 0.6 catches sustained trends without over-blocking.")

// Velocity filter
velLen       = input.int(9,    "Velocity Smoothing",          group="Velocity Filter")
maxVelMult   = input.float(0.9,"Max Velocity (ATR multiples)",group="Velocity Filter", step=0.1)

// ATR
atrLen       = input.int(13,   "ATR Length",                  group="ATR")
atrMult      = input.float(2.2,"ATR Trail Stop Multiplier",   group="ATR", step=0.1)

// Cooldown / Session / Direction
cooldown     = input.int(6,    "Bars Between Trades",         group="Filters")
useSession   = input.bool(true,"MCX Session Filter",          group="Session")
sessStart    = input.int(920,  "Start HHMM",                  group="Session")
sessEnd      = input.int(2300, "End HHMM",                    group="Session")
tradeDir     = input.string("Both","Direction",
     options=["Long Only","Short Only","Both"],                group="Filters")

// ═══════════════════════════════════════════════════════
// SESSION
// ═══════════════════════════════════════════════════════
hhmm      = hour * 100 + minute
inSession = not useSession or (hhmm >= sessStart and hhmm <= sessEnd)

// ═══════════════════════════════════════════════════════
// KALMAN FILTER — unchanged
// ═══════════════════════════════════════════════════════
var float kalmX  = close
var float kalmV  = 0.0
var float kalmP  = 1.0

kalmXp = kalmX + kalmV
kalmPp = kalmP + kalmQ
kalmK  = kalmPp / (kalmPp + kalmR)

kalmX := kalmXp + kalmK * (close - kalmXp)
kalmV := kalmV  + kalmK * (close - kalmXp) * 0.1
kalmP := (1 - kalmK) * kalmPp

// ═══════════════════════════════════════════════════════
// ATR-NORMALISED RESIDUAL — unchanged
// ═══════════════════════════════════════════════════════
atr       = ta.atr(atrLen)
residual  = close - kalmX
normResid = atr > 0 ? residual / atr : 0.0

// ═══════════════════════════════════════════════════════
// GAP CIRCUIT BREAKER — unchanged
// ═══════════════════════════════════════════════════════
prevClose  = nz(close[1], close)
gapSize    = math.abs(open - prevClose)
barRange   = high - low

gapTriggered   = atr > 0 and gapSize  > gapThresh   * atr
rangeTriggered = atr > 0 and barRange > rangeThresh * atr
anyGap         = gapTriggered or rangeTriggered

var int barsSinceGap = lockoutBars + 1
barsSinceGap := anyGap ? 0 : math.min(barsSinceGap + 1, lockoutBars + 1)
gapOK = not useGapBreaker or barsSinceGap >= lockoutBars

// ═══════════════════════════════════════════════════════
// OU HALF-LIFE GATE — unchanged
// ═══════════════════════════════════════════════════════
resLag   = residual[1]
resDelta = residual - resLag

meanX    = ta.sma(resLag,   ouLen)
meanDX   = ta.sma(resDelta, ouLen)
covXDX   = ta.sma((resLag - meanX) * (resDelta - meanDX), ouLen)
varX     = ta.variance(resLag, ouLen)

beta     = varX > 0 ? covXDX / varX : 0.0
kappa    = -beta
halfLife = kappa > 0 ? math.log(2) / kappa : 9999.0
ouOK     = not useOuGate or (kappa > 0 and halfLife <= maxHalfLife)

// ═══════════════════════════════════════════════════════
// VELOCITY FILTER — unchanged
// ═══════════════════════════════════════════════════════
kalmVelAbs    = math.abs(kalmV)
kalmVelSmooth = ta.ema(kalmVelAbs, velLen)
kalmVelNorm   = atr > 0 ? kalmVelSmooth / atr : 0.0
velOK         = kalmVelNorm < maxVelMult

// ═══════════════════════════════════════════════════════
// VARIANCE RATIO — unchanged
// ═══════════════════════════════════════════════════════
r1      = close - close[1]
rN      = close - close[vrLen]
var1    = ta.variance(r1, vrLen)
varN    = ta.variance(rN, vrLen)
vr      = (var1 > 0) ? varN / (vrLen * var1) : 1.0
vrMR    = vr < vrThreshold

// ═══════════════════════════════════════════════════════
// HURST R/S PROXY — NEW
// Computes rescaled range over the last hurstLen bars.
// R/S relationship: R/S ∝ n^H where H is the Hurst exponent.
// Single-scale proxy: H ≈ log(R/S) / log(n)
// This is the Peters (1994) simplified form — not the full
// multi-scale Mandelbrot R/S, but tracks the full version
// closely in practice for real-time filtering.
//
// Steps:
//   1. Compute log-returns over window
//   2. Build cumulative deviation from mean
//   3. R = max(cumdev) - min(cumdev)
//   4. S = stdev of returns
//   5. H = log(R/S) / log(n)
//
// Interpretation:
//   H < 0.5 → anti-persistent (mean-reverting)
//   H ≈ 0.5 → random walk
//   H > 0.5 → persistent (trending)
//
// Gate: block entries when H > hurstMaxH (extreme trending only)
// ═══════════════════════════════════════════════════════

// Build log-return series, then cumulative deviation from mean,
// then compute R (range of cumdev) and S (stdev of returns).
// In Pine, we iterate over the lookback on every bar — acceptable
// at hurstLen = 64.

hurstH = 0.5  // default to random walk when insufficient data

if bar_index >= hurstLen
    // 1. Mean log-return over window
    float sumRet = 0.0
    for i = 0 to hurstLen - 1
        ret = math.log(close[i] / close[i + 1])
        sumRet += ret
    meanRet = sumRet / hurstLen

    // 2. Compute cumulative deviation series and track min/max
    //    Also accumulate sum-of-squared-deviations for S
    float cumDev    = 0.0
    float cumDevMin = 0.0
    float cumDevMax = 0.0
    float sumSqDev  = 0.0
    for i = 0 to hurstLen - 1
        ret = math.log(close[hurstLen - 1 - i] / close[hurstLen - i])
        dev = ret - meanRet
        cumDev += dev
        cumDevMin := math.min(cumDevMin, cumDev)
        cumDevMax := math.max(cumDevMax, cumDev)
        sumSqDev  += dev * dev

    // 3. R = range of cumulative deviation
    R = cumDevMax - cumDevMin

    // 4. S = stdev of returns (population form)
    S = math.sqrt(sumSqDev / hurstLen)

    // 5. H = log(R/S) / log(n)
    //    Clamp R/S to avoid log(0); minimum meaningful R/S is ~1
    if S > 0 and R > 0
        rs = R / S
        if rs > 1.0
            hurstH := math.log(rs) / math.log(hurstLen)
        else
            hurstH := 0.5  // degenerate case, treat as random walk

// Gate: block only when Hurst indicates EXTREME trending
// (not just any H > 0.5 — that would overlap too much with VR)
hurstOK = not useHurstGate or hurstH <= hurstMaxH

// ═══════════════════════════════════════════════════════
// COOLDOWN
// ═══════════════════════════════════════════════════════
var int barsSince = cooldown + 1
barsSince := strategy.position_size != 0 ? 0 : math.min(barsSince + 1, cooldown + 1)
cooledDown = barsSince >= cooldown

// ═══════════════════════════════════════════════════════
// ENTRIES — now with six gates (added hurstOK)
// ═══════════════════════════════════════════════════════
longCond  = normResid < -entryThresh and vrMR and velOK and ouOK and gapOK and hurstOK and inSession and cooledDown
shortCond = normResid >  entryThresh and vrMR and velOK and ouOK and gapOK and hurstOK and inSession and cooledDown

goLong  = longCond  and (tradeDir == "Long Only"  or tradeDir == "Both")
goShort = shortCond and (tradeDir == "Short Only" or tradeDir == "Both")

// ═══════════════════════════════════════════════════════
// EXITS — unchanged
// ═══════════════════════════════════════════════════════
longExit  = normResid >= -exitThresh
shortExit = normResid <=  exitThresh

var float trailL = na
var float trailS = na

if strategy.position_size > 0
    trailL := math.max(nz(trailL, low - atr * atrMult), close - atr * atrMult)
if strategy.position_size < 0
    trailS := math.min(nz(trailS, high + atr * atrMult), close + atr * atrMult)
if strategy.position_size == 0
    trailL := na
    trailS := na

// ═══════════════════════════════════════════════════════
// EXECUTION — unchanged
// ═══════════════════════════════════════════════════════
if goLong  and strategy.position_size == 0
    strategy.entry("Long",  strategy.long)
if goShort and strategy.position_size == 0
    strategy.entry("Short", strategy.short)

if strategy.position_size > 0
    strategy.exit("L-Trail", "Long",  stop=trailL)
if strategy.position_size < 0
    strategy.exit("S-Trail", "Short", stop=trailS)

if strategy.position_size > 0 and longExit
    strategy.close("Long",  comment="Revert")
if strategy.position_size < 0 and shortExit
    strategy.close("Short", comment="Revert")

// ═══════════════════════════════════════════════════════
// PLOTS
// ═══════════════════════════════════════════════════════
residCol = normResid >  entryThresh ? color.new(color.red,  10) :
           normResid < -entryThresh ? color.new(color.lime, 10) :
                                      color.new(color.gray, 50)
plot(normResid, "Norm Residual", color=residCol, linewidth=2)

vrCol = vr < vrThreshold ? color.new(color.aqua, 30) :        vr > 1.0         ? color.new(color.orange, 30) :                           color.new(color.gray, 50)
plot(vr, "Variance Ratio", color=vrCol, linewidth=1)

plot(kalmVelNorm, "Kalman Velocity", color=color.new(color.purple, 40), linewidth=1)

hlPlot = halfLife > 50 ? na : halfLife
plot(hlPlot, "OU Half-Life", color=color.new(color.yellow, 30), linewidth=1)

// NEW: Hurst plot — orange when extreme trending, green when mean-reverting
hurstCol = hurstH > hurstMaxH ? color.new(color.orange, 20) :           hurstH < 0.5       ? color.new(color.green,  30) :                                color.new(color.gray,   50)
plot(hurstH, "Hurst H", color=hurstCol, linewidth=1)

hline( entryThresh, "Entry Short",   color=color.new(color.red,  30), linestyle=hline.style_solid)
hline(-entryThresh, "Entry Long",    color=color.new(color.lime, 30), linestyle=hline.style_solid)
hline( exitThresh,  "Exit Short",    color=color.new(color.red,  60), linestyle=hline.style_dashed)
hline(-exitThresh,  "Exit Long",     color=color.new(color.lime, 60), linestyle=hline.style_dashed)
hline(vrThreshold,  "VR Threshold",  color=color.new(color.aqua, 50), linestyle=hline.style_dotted)
hline(1.0,          "VR=1 (RW)",     color=color.new(color.gray, 40), linestyle=hline.style_dotted)
hline(0,            "Zero",          color=color.new(color.white,60), linestyle=hline.style_solid)

// NEW: Hurst reference lines
hline(hurstMaxH,    "Hurst Block",   color=color.new(color.orange,30), linestyle=hline.style_dotted)
hline(0.5,          "H=0.5 (RW)",    color=color.new(color.gray,  40), linestyle=hline.style_dotted)

bgcolor(useGapBreaker and not gapOK ? color.new(color.red, 85) :        vrMR and velOK and ouOK and hurstOK ? color.new(color.aqua, 94) :
                                              color.new(color.orange, 94),
     title="Regime BG (Red=gap lockout, Aqua=all pass, Orange=trend)")
